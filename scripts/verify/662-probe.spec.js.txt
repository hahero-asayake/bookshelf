// 記録用 (イシュー#662 step2・commit しない。写しを scripts/verify/662-probe.spec.js.txt に置く)
// E (手動追加→待たずに再読込) の後に、次の保存を起こして4冊目が落ちるかを測る。
import { readFileSync, appendFileSync, mkdirSync } from 'node:fs';
import { test, expect } from './helpers/test-base.js';
import { bootAppForTest } from './helpers/gate-setup.js';
import { stubFolderPickerForTest, connectLocalFolderForTest } from './helpers/gate-local.js';

const EXPORT_JSON = readFileSync(new URL('../fixtures/gate-kindle-export.json', import.meta.url), 'utf-8');
const OUT = '/tmp/i662/probe.jsonl';
const allShelf = (page) => page.locator('#sidebar-bookshelf-tree [data-bookshelf-id="all"]');
test.use({ viewport: { width: 1280, height: 800 } });

const snap = (page, dirName) => page.evaluate(async (name) => {
    const out = {};
    const rd = async (dir, f) => JSON.parse(await (await (await dir.getFileHandle(f)).getFile()).text());
    try {
        let dir = await (await navigator.storage.getDirectory()).getDirectoryHandle(name);
        dir = await dir.getDirectoryHandle('private');
        try { out.libN = ((await rd(dir, 'library.json')).books || []).length; } catch (e) { out.libN = null; }
        try { out.allN = ((await rd(await dir.getDirectoryHandle('bookshelves'), 'all.json')).books || []).length; } catch (e) { out.allN = null; }
        try { out.metaN = ((await rd(dir, 'bookshelves.json')).bookshelves || []).length; } catch (e) { out.metaN = null; }
    } catch (e) { out.err = String(e && e.name); }
    try {
        const u = JSON.parse(localStorage.getItem('virtualBookshelf_userData') || 'null');
        out.lsOrder = u.bookOrder.all.length;
        out.lsShelf = (u.bookshelves.find((b) => b.isSpecial) || {}).books?.length ?? null;
    } catch (e) { out.lsErr = true; }
    const ud = window.bookshelf && window.bookshelf.userData;
    if (ud) {
        const sh = (ud.bookshelves || []).find((b) => b.isSpecial);
        out.memShelf = sh && sh.books ? sh.books.length : null;
        out.memOrder = ud.bookOrder && ud.bookOrder.all ? ud.bookOrder.all.length : null;
        out.memSame = !!(sh && ud.bookOrder && sh.books === ud.bookOrder.all);
        out.books = (window.bookshelf.books || []).length;
    }
    out.pending = localStorage.getItem('virtualBookshelf_pendingSync');
    return out;
}, dirName);

async function settle(page) {
    await expect(allShelf(page)).toBeVisible({ timeout: 20000 });
    const counts = [];
    for (let i = 0; i < 12; i++) { counts.push(await allShelf(page).locator('.bsr-count').textContent().catch(() => '?')); await page.waitForTimeout(250); }
    const dialog = await page.locator('#load-conflict-dialog').filter({ visible: true }).count();
    return { counts: [...new Set(counts)].join('>'), dialog };
}

for (const choice of ['local', 'none']) {
    test(`662 probe loss ${choice}`, async ({ page }, testInfo) => {
        const dirName = `p662-${choice}-${testInfo.repeatEachIndex}`;
        const rec = { choice, n: testInfo.repeatEachIndex };
        await stubFolderPickerForTest(page, dirName);
        await bootAppForTest(page);
        await connectLocalFolderForTest(page);
        rec.sameAfterConnect = (await snap(page, dirName)).memSame;
        await page.locator('#dashboard-welcome button[data-dw="import"]').click();
        await page.locator('[data-import-action="open-paste"]').click();
        await page.locator('#kindle-paste-input').fill(EXPORT_JSON);
        await page.locator('#import-from-paste').click();
        await page.locator('#select-all-books').click();
        await page.locator('#import-selected-books').click();
        await expect(page.locator('#import-results')).toContainText('取込結果');
        rec.afterImport = await snap(page, dirName);
        await expect.poll(async () => (await snap(page, dirName)).allN, { timeout: 20000 }).toBe(3);
        await page.waitForTimeout(500);
        rec.afterFirstWrite = await snap(page, dirName);
        await page.keyboard.press('Escape');
        await page.locator('#sidebar-settings').click();
        await page.locator('#settings-master .settings-cat[data-cat="library-section"]').click();
        await page.locator('#add-book-manually').click();
        await page.locator('#manual-asin').fill('B0M662ADD1');
        await page.locator('#manual-title').fill('手動追加の本');
        await page.locator('#add-manually').click();
        await expect(page.locator('#add-book-results')).toContainText('書籍を追加しました');
        rec.afterAdd = await snap(page, dirName);
        if (choice === 'none') {
            // F と同じく待つ：localStorage の「すべて」本棚が書き直されるか (再読込しない)
            const t0 = Date.now();
            await expect.poll(async () => (await snap(page, dirName)).lsShelf, { timeout: 12000 }).toBe(4).catch(() => {});
            rec.waitNoReload = { ms: Date.now() - t0, ...(await snap(page, dirName)) };
            return appendFileSync(OUT, JSON.stringify(rec) + '\n');
        }
        await page.reload();
        rec.reload1 = { ...(await settle(page)), ...(await snap(page, dirName)) };
        if (rec.reload1.dialog) {
            await page.locator('#load-conflict-dialog .cfm-local').click();
            await page.waitForTimeout(300);
            rec.afterChooseLocal = await snap(page, dirName);
        }
        // 次の保存 (保存の入口 saveUserData を呼ぶ＝どの編集も最後に通る所)
        await page.evaluate(() => window.bookshelf.saveUserData());
        await expect.poll(async () => (await snap(page, dirName)).pending, { timeout: 15000 }).toBe(null).catch(() => {});
        await page.waitForTimeout(1000);
        rec.afterNextSave = await snap(page, dirName);
        await page.reload();
        rec.reload2 = { ...(await settle(page)), ...(await snap(page, dirName)) };
        mkdirSync('/tmp/i662', { recursive: true });
        appendFileSync(OUT, JSON.stringify(rec) + '\n');
    });
}
